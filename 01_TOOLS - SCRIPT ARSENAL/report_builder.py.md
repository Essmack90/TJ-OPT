#!/usr/bin/env python3
"""
report_builder.py - OSCP report scaffolder
Usage: python3 report_builder.py -t 10.10.10.5 --name "Box Name"
"""

import argparse
import json
import os
from datetime import datetime

R = '\033[91m'
G = '\033[92m'
Y = '\033[93m'
C = '\033[96m'
B = '\033[1m'
X = '\033[0m'

def good(s): print(f"{G}[+]{X} {s}")
def info(s): print(f"{C}[*]{X} {s}")

REPORT_TEMPLATE = """# {box_name} — Penetration Test Report

**Target IP:** {target}
**Date:** {date}
**Tester:** {tester}
**Status:** {status}

---

## Executive Summary

{box_name} ({target}) was compromised via {attack_summary}. Initial access was
obtained through {initial_access}. Privilege escalation was achieved by
{privesc_method}.

---

## Enumeration

### Open Ports / Services

{services_table}

### Key Findings from Enumeration

{enum_findings}

---

## Exploitation

### Vulnerability

{vuln_description}

**CVE:** {cve}
**CVSS:** {cvss}

### Proof of Concept


**Screenshot placeholder:** [insert screenshot of initial shell]

---

## Post-Exploitation

### Local Enumeration

{post_exploit_notes}

### Privilege Escalation

**Method:** {privesc_method}


**Screenshot placeholder:** [insert screenshot of root/SYSTEM shell]

---

## Loot

### Flags

| Flag      | Value     | Location  |
|-----------|-----------|-----------|
| user.txt  | {user_flag}  | {user_flag_path} |
| root.txt  | {root_flag}  | {root_flag_path} |

### Credentials Found

{creds_table}

---

## Remediation Recommendations

{remediation}

---

## Appendix — Raw Tool Output

### nmap


### enum4linux


---
*Report generated by report_builder.py on {date}*
"""

def load_json(path):
    if not path or not os.path.exists(path):
        return {}
    try:
        with open(path) as f:
            return json.load(f)
    except Exception:
        return {}

def build_services_table(enum_data):
    services = enum_data.get("findings", {}).get("shares", [])
    if not services:
        return "| Port | Service | Version | Notes |\n|------|---------|---------|-------|\n| ?    | ?       | ?       | Manual review required |"
    lines = ["| Port | Service | Version | Notes |", "|------|---------|---------|-------|"]
    for s in services:
        lines.append(f"| ?    | {s.get('name','?')} | ?  | {s.get('comment','')} |")
    return "\n".join(lines)

def build_creds_table(loot_data):
    creds = loot_data.get("loot", {}).get("plaintext", []) + loot_data.get("loot", {}).get("config_creds", [])
    if not creds:
        return "| Service | Username | Password | Source |\n|---------|----------|----------|--------|\n| —       | —        | —        | —      |"
    lines = ["| Service | Username | Password | Source |", "|---------|----------|----------|--------|"]
    for c in creds:
        svc = c.get("service", c.get("type", "?"))
        user = c.get("user", "?")
        pw = c.get("pass", c.get("value", "?"))
        src = c.get("file", "?")
        lines.append(f"| {svc} | {user} | {pw} | {src} |")
    return "\n".join(lines)

def build_enum_findings(enum_data):
    findings = enum_data.get("findings", {})
    lines = []
    if findings.get("anon_smb"):
        lines.append("- **Anonymous SMB access** — shares browsable without credentials")
    if findings.get("anon_rpc"):
        lines.append("- **Anonymous RPC access** — domain information leaked")
    users = findings.get("users", [])
    if users:
        unames = ", ".join(u["name"] for u in users)
        lines.append(f"- **Users enumerated via RPC:** {unames}")
    shares = findings.get("shares", [])
    if shares:
        snames = ", ".join(s["name"] for s in shares)
        lines.append(f"- **SMB shares discovered:** {snames}")
    pp = findings.get("password_policy", {})
    if pp:
        lines.append(f"- **Password policy:** min length {pp.get('min_length','?')}, lockout {pp.get('lockout_threshold','?')}")
    return "\n".join(lines) if lines else "Manual review required — no JSON input provided."

def build_remediation(enum_data, loot_data):
    recs = []
    findings = enum_data.get("findings", {})
    if findings.get("anon_smb"):
        recs.append("- Disable anonymous SMB access. Enforce authentication on all shares.")
    if findings.get("anon_rpc"):
        recs.append("- Restrict anonymous RPC access via `RestrictAnonymous = 2` in the registry.")
    if loot_data.get("loot", {}).get("hashes"):
        recs.append("- Enforce strong password policy. Rotate all compromised credentials immediately.")
    if loot_data.get("loot", {}).get("ssh_keys"):
        recs.append("- Revoke and regenerate exposed SSH keys. Audit authorized_keys files.")
    if not recs:
        recs.append("- Review all services for unnecessary exposure.")
        recs.append("- Apply principle of least privilege to all service accounts.")
        recs.append("- Ensure all software is patched to current versions.")
    return "\n".join(recs)

def find_flags(loot_data):
    user_flag = "**REDACTED**"
    root_flag = "**REDACTED**"
    user_path = "?"
    root_path = "?"
    for item in loot_data.get("loot", {}).get("interesting", []):
        path = item if isinstance(item, str) else item.get("path", "")
        if "user.txt" in path:
            user_path = path
        if "root.txt" in path or "proof.txt" in path:
            root_path = path
    return user_flag, user_path, root_flag, root_path

def main():
    ap = argparse.ArgumentParser(description="report_builder.py — OSCP report scaffolder")
    ap.add_argument("-t", "--target", required=True)
    ap.add_argument("--name", default=None, help="Box name")
    ap.add_argument("--tester", default="Your Name", help="Your name")
    ap.add_argument("--enum", default=None, help="post_enum JSON report path")
    ap.add_argument("--loot", default=None, help="loot_parser JSON output path")
    ap.add_argument("--status", default="Compromised — root/SYSTEM obtained")
    ap.add_argument("--output", default=".", help="Output directory")
    ap.add_argument("--cve", default="TBD")
    ap.add_argument("--cvss", default="TBD")
    ap.add_argument("--poc", default="# Add PoC commands here")
    ap.add_argument("--privesc", default="TBD — fill in method")
    ap.add_argument("--privesc-cmds", default="# Add privesc commands here")
    ap.add_argument("--initial-access", default="TBD — describe initial foothold")
    ap.add_argument("--attack-summary", default="a chain of enumeration and service exploitation")
    args = ap.parse_args()

    box_name = args.name or f"Box-{args.target.replace('.', '_')}"
    enum_data = load_json(args.enum)
    loot_data = load_json(args.loot)

    user_flag, user_path, root_flag, root_path = find_flags(loot_data)
    services_table = build_services_table(enum_data)
    enum_findings = build_enum_findings(enum_data)
    creds_table = build_creds_table(loot_data)
    remediation = build_remediation(enum_data, loot_data)

    post_notes = []
    hashes = loot_data.get("loot", {}).get("hashes", [])
    if hashes:
        post_notes.append(f"- {len(hashes)} password hash(es) extracted from /etc/shadow")
    ssh_keys = loot_data.get("loot", {}).get("ssh_keys", [])
    if ssh_keys:
        post_notes.append(f"- {len(ssh_keys)} SSH private key(s) found")
    histories = loot_data.get("loot", {}).get("history", [])
    if histories:
        post_notes.append(f"- {len(histories)} interesting history line(s) found")
    if not post_notes:
        post_notes.append("- Manual post-exploitation — add notes here")

    report = REPORT_TEMPLATE.format(
        box_name=box_name,
        target=args.target,
        date=datetime.now().strftime("%Y-%m-%d"),
        tester=args.tester,
        status=args.status,
        attack_summary=args.attack_summary,
        initial_access=args.initial_access,
        services_table=services_table,
        enum_findings=enum_findings,
        vuln_description="[Describe the vulnerability exploited]",
        cve=args.cve,
        cvss=args.cvss,
        poc_commands=args.poc,
        post_exploit_notes="\n".join(post_notes),
        privesc_method=args.privesc,
        privesc_commands=args.privesc_cmds,
        user_flag=user_flag,
        user_flag_path=user_path,
        root_flag=root_flag,
        root_flag_path=root_path,
        creds_table=creds_table,
        remediation=remediation,
        nmap_output="[Paste nmap output here]",
        enum4linux_output="[Paste enum4linux output here]",
    )

    os.makedirs(args.output, exist_ok=True)
    fname = os.path.join(args.output, f"{box_name.replace(' ', '_')}_report.md")
    with open(fname, "w") as f:
        f.write(report)

    good(f"Report scaffolded: {fname}")
    info("Fill in the blanks — especially PoC steps, screenshots, and flags")
    info("Convert to PDF/DOCX with: pandoc report.md -o report.pdf")

if __name__ == "__main__":
    main()
